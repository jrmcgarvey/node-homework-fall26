SELECT o.order_id, SUM(p.price * li.quantity) total_price  FROM orders o JOIN line_items li ON o.order_id = li.order_id JOIN products p ON p.product_id = li.product_id GROUP BY o.order_id ORDER BY o.order_id LIMIT 5;
SELECT c.customer_id, c.customer_name, AVG(total_price) average_order_price FROM customers c JOIN (SELECT o.customer_id, SUM(p.price * li.quantity) total_price  FROM orders o JOIN line_items li ON o.order_id = li.order_id JOIN products p ON p.product_id = li.product_id GROUP BY o.order_id) sub ON sub.customer_id = c.customer_id GROUP BY c.customer_id, c.customer_name ORDER BY c.customer_name;
